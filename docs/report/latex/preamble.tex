% --- Hình đẳng cự trên bìa: bàn ô và khối 1×1×2 đứng --------------------------
% \isobox{x0}{y0}{z0}{x1}{y1}{z1}{màu}: hộp thấy ba mặt (trên, phải, trái).
\newcommand{\isobox}[7]{%
  \fill[#7] (#1,#2,#6)--(#4,#2,#6)--(#4,#5,#6)--(#1,#5,#6)--cycle;
  \fill[#7!78!black] (#4,#2,#3)--(#4,#5,#3)--(#4,#5,#6)--(#4,#2,#6)--cycle;
  \fill[#7!60!black] (#1,#5,#3)--(#4,#5,#3)--(#4,#5,#6)--(#1,#5,#6)--cycle;}
\newcommand{\isotile}[3]{% cột i, hàng j, màu mặt trên
  \pgfmathsetmacro\xa{#1+0.05}\pgfmathsetmacro\xb{#1+0.95}%
  \pgfmathsetmacro\ya{#2+0.05}\pgfmathsetmacro\yb{#2+0.95}%
  \isobox{\xa}{\ya}{0}{\xb}{\yb}{0.28}{#3}}
\newcommand{\isoscene}{%

    \foreach \i/\j/\c in {0/0/isofloor, 1/0/isofloor, 0/1/isofloor, 2/0/isofloor,

        5/1/isofloor, 4/2/isofloor, 3/3/isofloor, 6/1/isofloor, 5/2/isogoal,
        4/3/isofloor, 6/2/isofloor, 5/3/isofloor}
